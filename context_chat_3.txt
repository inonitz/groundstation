    int getch() {

        int ch;

        struct termios oldt, newt;

        tcgetattr(STDIN_FILENO, &oldt);

        newt = oldt;

        newt.c_lflag &= ~(ICANON | ECHO);

        tcsetattr(STDIN_FILENO, TCSANOW, &newt);

        ch = getchar();

        tcsetattr(STDIN_FILENO, TCSANOW, &oldt);

        return ch;

    }


    void key_loop() {

        while (rclcpp::ok() && running_) {

            int c = getch();

            std_msgs::msg::String msg;

            switch(c) {

                case 't': msg.data = "takeoff"; break;

                case 'l': msg.data = "land"; break;

                case 'w': msg.data = "rc 0 50 0 0"; break;  // Forward

                case 's': msg.data = "rc 0 -50 0 0"; break; // Back

                case 'a': msg.data = "rc -50 0 0 0"; break; // Left

                case 'd': msg.data = "rc 50 0 0 0"; break;  // Right

                case ' ': msg.data = "rc 0 0 0 0"; break;   // Stop/Hover

                case 'q': rclcpp::shutdown(); return;

                default: continue;

            }

            cmd_pub_->publish(msg);

        }

    }

void publish_odometry(const TelloState& state) {
    auto current_time = this->now();
    
    // Calculate dt for integration
    double dt = 0.0;
    if (last_time_.nanoseconds() > 0) {
        dt = (current_time - last_time_).seconds();
    }
    last_time_ = current_time;

    // Convert dm/s to m/s
    double vx = state.vgx / 10.0;
    double vy = state.vgy / 10.0;
    
    // Integrate position X, Y
    pos_x_ += vx * dt;
    pos_y_ += vy * dt;

    nav_msgs::msg::Odometry odom;
    odom.header.stamp = current_time;
    odom.header.frame_id = "odom";
    odom.child_frame_id = "base_link";

    // Position (convert h cm to meters)
    odom.pose.pose.position.x = pos_x_;
    odom.pose.pose.position.y = pos_y_;
    odom.pose.pose.position.z = state.h / 100.0;

    // Orientation (Convert Euler degrees to Quaternion)
    tf2::Quaternion q;
    q.setRPY(state.roll * M_PI / 180.0, state.pitch * M_PI / 180.0, state.yaw * M_PI / 180.0);
    odom.pose.pose.orientation.x = q.x();
    odom.pose.pose.orientation.y = q.y();
    odom.pose.pose.orientation.z = q.z();
    odom.pose.pose.orientation.w = q.w();

    // Linear Velocity
    odom.twist.twist.linear.x = vx;
    odom.twist.twist.linear.y = vy;
    odom.twist.twist.linear.z = state.vgz / 10.0;

    odom_pub_->publish(odom);
}
